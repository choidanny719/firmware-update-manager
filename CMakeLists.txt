cmake_minimum_required(VERSION 3.24)
project(firmware_update_manager VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(BUILD_RUNTIME "Build the FreeRTOS POSIX application" ON)
if(ENABLE_SANITIZERS AND BUILD_RUNTIME)
    message(FATAL_ERROR "Use -DBUILD_RUNTIME=OFF for sanitizer builds")
endif()
include(CTest)
include(FetchContent)

add_library(project_options INTERFACE)
target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined -no-pie)
endif()

add_library(firmware_core src/flash.c src/checksum.c src/result.c src/journal.c src/update.c src/packet.c)
target_include_directories(firmware_core PUBLIC include)
target_link_libraries(firmware_core PUBLIC project_options)
add_library(flash_sim sim/nor.c sim/flash_file.c)
target_include_directories(flash_sim PUBLIC sim)
target_compile_definitions(flash_sim PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(flash_sim PUBLIC firmware_core)

if(BUILD_TESTING)
    FetchContent_Declare(unity
        GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git
        GIT_TAG cbcd08fa7de711053a3deec6339ee89cad5d2697)
    FetchContent_MakeAvailable(unity)
    foreach(suite flash update packet)
        add_executable(test_${suite} tests/test_${suite}.c)
        target_link_libraries(test_${suite} PRIVATE flash_sim unity)
        add_test(NAME ${suite} COMMAND test_${suite})
    endforeach()
endif()

if(BUILD_RUNTIME)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The host simulator requires Linux; use WSL on Windows")
    endif()
    add_library(freertos_config INTERFACE)
    target_include_directories(freertos_config SYSTEM INTERFACE app)
    set(FREERTOS_PORT GCC_POSIX CACHE STRING "FreeRTOS port")
    set(FREERTOS_HEAP 4 CACHE STRING "FreeRTOS heap implementation")
    FetchContent_Declare(freertos_kernel
        GIT_REPOSITORY https://github.com/FreeRTOS/FreeRTOS-Kernel.git
        GIT_TAG 0adc196d4bd52a2d91102b525b0aafc1e14a2386)
    FetchContent_MakeAvailable(freertos_kernel)
    target_compile_definitions(freertos_kernel_port PRIVATE _POSIX_C_SOURCE=200809L)
    add_executable(fw_update app/main.c app/runtime.c)
    target_link_libraries(fw_update PRIVATE flash_sim freertos_kernel)
    if(BUILD_TESTING)
        find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
        add_test(NAME runtime_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE:fw_update>)
        set_tests_properties(runtime_smoke PROPERTIES TIMEOUT 20)
    endif()
endif()
